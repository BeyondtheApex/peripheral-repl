cmake_minimum_required(VERSION 3.24)
project(PeripheralNativeSurfaces LANGUAGES CXX)
set(native_sources taskbar_hook.cpp taskbar_loader.cpp xaml_host.cpp taskbar_trace.h taskbar_layout.h taskbar_geometry.h taskbar_placement.h taskbar_monitor.h taskbar_candidates.h taskbar_roots.h host.manifest CMakeLists.txt)
set(native_hashes "")
foreach(source IN LISTS native_sources)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" source_hash)
  string(APPEND native_hashes "${source_hash}")
endforeach()
string(SHA256 native_build "${native_hashes}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/native_build.h" "#pragma once\n#define PERIPHERAL_NATIVE_ABI 2\n#define PERIPHERAL_NATIVE_BUILD \"${native_build}\"\n")
include_directories("${CMAKE_CURRENT_BINARY_DIR}")
add_executable(peripheral-xaml-host WIN32 xaml_host.cpp)
target_compile_features(peripheral-xaml-host PRIVATE cxx_std_20)
target_compile_definitions(peripheral-xaml-host PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(peripheral-xaml-host PRIVATE /utf-8 /EHsc)
target_link_libraries(peripheral-xaml-host PRIVATE windowsapp d3d11 dxgi user32 ole32 shell32)
target_link_options(peripheral-xaml-host PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/host.manifest")
add_library(peripheral-taskbar-hook SHARED taskbar_hook.cpp taskbar_loader.cpp)
target_compile_features(peripheral-taskbar-hook PRIVATE cxx_std_20)
target_compile_definitions(peripheral-taskbar-hook PRIVATE UNICODE _UNICODE NOMINMAX PERIPHERAL_LIBRARY)
target_compile_options(peripheral-taskbar-hook PRIVATE /utf-8 /EHsc)
target_link_libraries(peripheral-taskbar-hook PRIVATE windowsapp d3d11 dxgi user32 ole32 shell32 advapi32)
set_property(TARGET peripheral-taskbar-hook PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
add_executable(peripheral-taskbar-loader taskbar_loader.cpp)
target_compile_features(peripheral-taskbar-loader PRIVATE cxx_std_20)
target_compile_definitions(peripheral-taskbar-loader PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(peripheral-taskbar-loader PRIVATE /utf-8 /EHsc)
target_link_libraries(peripheral-taskbar-loader PRIVATE ole32)
enable_testing()
add_executable(peripheral-taskbar-layout-tests taskbar_layout_tests.cpp)
target_compile_features(peripheral-taskbar-layout-tests PRIVATE cxx_std_20)
target_compile_definitions(peripheral-taskbar-layout-tests PRIVATE NOMINMAX)
add_test(NAME taskbar-layout COMMAND peripheral-taskbar-layout-tests)
add_executable(peripheral-taskbar-monitor-tests taskbar_monitor_tests.cpp)
target_compile_features(peripheral-taskbar-monitor-tests PRIVATE cxx_std_20)
target_compile_definitions(peripheral-taskbar-monitor-tests PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(peripheral-taskbar-monitor-tests PRIVATE user32)
add_test(NAME taskbar-monitor COMMAND peripheral-taskbar-monitor-tests)
add_executable(peripheral-taskbar-xaml-tests taskbar_xaml_tests.cpp)
target_compile_features(peripheral-taskbar-xaml-tests PRIVATE cxx_std_20)
target_compile_definitions(peripheral-taskbar-xaml-tests PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(peripheral-taskbar-xaml-tests PRIVATE /utf-8 /EHsc)
target_link_libraries(peripheral-taskbar-xaml-tests PRIVATE windowsapp d3d11 dxgi user32 ole32 shell32 advapi32)
target_link_options(peripheral-taskbar-xaml-tests PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/host.manifest")
# XAML islands need a desktop session; run this fixture locally, not on headless CI.
add_executable(peripheral-taskbar-diagnostics-tests taskbar_diagnostics_tests.cpp)
target_compile_features(peripheral-taskbar-diagnostics-tests PRIVATE cxx_std_20)
target_compile_definitions(peripheral-taskbar-diagnostics-tests PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(peripheral-taskbar-diagnostics-tests PRIVATE /utf-8 /EHsc)
target_link_libraries(peripheral-taskbar-diagnostics-tests PRIVATE windowsapp d3d11 dxgi user32 ole32 shell32 advapi32)
target_link_options(peripheral-taskbar-diagnostics-tests PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/host.manifest")
add_dependencies(peripheral-taskbar-diagnostics-tests peripheral-taskbar-hook)
