cmake_minimum_required(VERSION 3.24)
project(PeripheralNativeSurfaces LANGUAGES CXX)
set(native_sources taskbar_hook.cpp taskbar_loader.cpp xaml_host.cpp taskbar_trace.h host.manifest CMakeLists.txt)
set(native_hashes "")
foreach(source IN LISTS native_sources)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${source}" source_hash)
  string(APPEND native_hashes "${source_hash}")
endforeach()
string(SHA256 native_build "${native_hashes}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/native_build.h" "#pragma once\n#define PERIPHERAL_NATIVE_ABI 2\n#define PERIPHERAL_NATIVE_BUILD \"${native_build}\"\n")
include_directories("${CMAKE_CURRENT_BINARY_DIR}")
add_executable(peripheral-xaml-host WIN32 xaml_host.cpp)
target_compile_features(peripheral-xaml-host PRIVATE cxx_std_20)
target_compile_definitions(peripheral-xaml-host PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(peripheral-xaml-host PRIVATE /utf-8 /EHsc)
target_link_libraries(peripheral-xaml-host PRIVATE windowsapp d3d11 dxgi user32 ole32 shell32)
target_link_options(peripheral-xaml-host PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/host.manifest")
add_library(peripheral-taskbar-hook SHARED taskbar_hook.cpp taskbar_loader.cpp)
target_compile_features(peripheral-taskbar-hook PRIVATE cxx_std_20)
target_compile_definitions(peripheral-taskbar-hook PRIVATE UNICODE _UNICODE NOMINMAX PERIPHERAL_LIBRARY)
target_compile_options(peripheral-taskbar-hook PRIVATE /utf-8 /EHsc)
target_link_libraries(peripheral-taskbar-hook PRIVATE windowsapp d3d11 dxgi user32 ole32 shell32)
set_property(TARGET peripheral-taskbar-hook PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
add_executable(peripheral-taskbar-loader taskbar_loader.cpp)
target_compile_features(peripheral-taskbar-loader PRIVATE cxx_std_20)
target_compile_definitions(peripheral-taskbar-loader PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(peripheral-taskbar-loader PRIVATE /utf-8 /EHsc)
target_link_libraries(peripheral-taskbar-loader PRIVATE ole32)
